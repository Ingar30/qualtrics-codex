\begin{frame}{Workflow}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.52\textwidth}
      \begin{enumerate}
        \item Write a survey specification in JSON.
        \item Create or connect to a Qualtrics survey.
        \item Export responses as CSV.
        \item Clean and summarize responses with Stata or Python.
        \item Build Beamer slides, with Python slides as fallback.
      \end{enumerate}
    \end{column}
    \begin{column}{0.42\textwidth}
      \begin{block}{Design principle}
        Keep the base path easy to inspect. Extensions should reuse the same survey key, data folders, and slide inputs.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{Analysis Outputs}
