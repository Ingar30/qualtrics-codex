\begin{frame}{Workflow}
  \begin{enumerate}
    \item Design a neutral public opinion survey.
    \item Submit synthetic responses through Qualtrics for the live demo.
    \item Export responses into an ignored raw folder.
    \item Clean with Stata when available, otherwise Python.
    \item Build figures and slides from reproducible local outputs.
  \end{enumerate}
\end{frame}

\begin{frame}{Survey Design}
  \begin{block}{Measures}
    Perceived prevalence, likely mechanisms, persuasive evidence, policy responses, and confidence.
  \end{block}
  Only synthetic/demo outputs are public by default.
\end{frame}
